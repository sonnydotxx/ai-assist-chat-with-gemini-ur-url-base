def fetch_knowledge_base(url):
    try:
        headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'}
        session = requests.Session()
        session.trust_env = False  
        
        response = session.get(url, headers=headers, timeout=5)
        response.encoding = 'utf-8'
        soup = BeautifulSoup(response.text, 'html.parser')
        
        for script in soup(["script", "style", "header", "footer", "nav"]):
            script.extract()
            
        text = soup.get_text(separator=' ', strip=True)
        print("   ✅ База знаний успешно загружена!")
        return text[:12000]
    except Exception as e:
        print(f"   ⚠️ Ошибка загрузки базы знаний: {e}")
        return "База знаний недоступна."

kb_content = fetch_knowledge_base(KNOWLEDGE_BASE_URL)

def get_verified_gemini_model(genai_client):
    print("   🔍 Поиск и проверка активных моделей...")
    candidates = ["gemini-1.5-flash", "gemini-1.5-flash-8b", "gemini-1.5-pro"]
    
    try:
        for m in genai_client.models.list():
            name = m.name.replace("models/", "") if hasattr(m, 'name') else str(m)
            if name not in candidates:
                candidates.append(name)
    except Exception:
        pass

    for model_id in candidates:
        try:
            genai_client.models.generate_content(model=model_id, contents="ping")
            print(f"   ✅ Подключена модель: '{model_id}'")
            return model_id
        except Exception:
            continue

    return "gemini-1.5-flash"

    def create_new_chat(self):
        self.chat_counter += 1
        chat_id = f"chat_{self.chat_counter}"
        title = f"Диалог #{self.chat_counter}"

        self.chats[chat_id] = {
            "title": title,
            "messages": {}
        }

        item = QListWidgetItem(f"💬 {title}")
        item.setData(Qt.ItemDataRole.UserRole, chat_id)
        self.chat_list_widget.addItem(item)
        
        self.chat_list_widget.setCurrentItem(item)
        self.switch_to_chat(chat_id)

    def on_chat_selected(self, item):
        chat_id = item.data(Qt.ItemDataRole.UserRole)
        self.switch_to_chat(chat_id)

    def switch_to_chat(self, chat_id):
        self.current_chat_id = chat_id
        chat_data = self.chats[chat_id]
        self.label_chat_title.setText(f"💬 {chat_data['title']}")
        self.render_chat_messages(chat_id)

    def send_request(self):
        text = self.input_field.text().strip()
        if not text or not self.current_chat_id:
            return

        self.input_field.clear()
        chat_id = self.current_chat_id
        chat_data = self.chats[chat_id]

        if len(chat_data["messages"]) == 0:
            new_title = text[:18] + "..." if len(text) > 18 else text
            chat_data["title"] = new_title
            self.label_chat_title.setText(f"💬 {new_title}")
            
            current_item = self.chat_list_widget.currentItem()
            if current_item:
                current_item.setText(f"💬 {new_title}")

        self.msg_counter += 1
        msg_id = f"msg_{self.msg_counter}"

        self.update_item_signal.emit(chat_id, msg_id, text, "⏳ Gemini формирует развернутый ответ...")

        threading.Thread(
            target=self.get_gemini_hint, 
            args=(chat_id, msg_id, text), 
            daemon=True
        ).start()

    def handle_message_update(self, chat_id, msg_id, client_text, hint_text):
        if chat_id not in self.chats:
            return

        time_str = datetime.now().strftime("%H:%M:%S")
        self.chats[chat_id]["messages"][msg_id] = {
            "time": time_str,
            "client": client_text,
            "hint": hint_text
        }

        if self.current_chat_id == chat_id:
            self.render_chat_messages(chat_id)

    def render_chat_messages(self, chat_id):
        chat_data = self.chats.get(chat_id, {})
        messages = chat_data.get("messages", {})

        if not messages:
            self.text_display.setHtml("")
            return

        html_blocks = []
        for m_id, item in messages.items():
            if "⏳" in item['hint']:
                hint_style = "color: #FBBF24; font-style: italic;"
            elif "Ошибка" in item['hint']:
                hint_style = "color: #F87171;"
            else:
                hint_style = "color: #34D399; line-height: 1.4;"

            card_html = f"""
            <div style="
                background: rgba(30, 41, 59, 0.75); 
                border: 1px solid rgba(249, 115, 22, 0.15); 
                border-radius: 10px; 
                padding: 10px 14px; 
                margin-bottom: 12px;">
                
                <div style="margin-bottom: 6px;">
                    <span style="color: #64748B; font-size: 11px; font-weight: 600;">[{item['time']}]</span>
                    <span style="color: #F97316; font-weight: 700; font-size: 13px; margin-left: 6px;">Запрос:</span>
                    <span style="color: #F1F5F9; font-size: 13px; margin-left: 4px;">{item['client']}</span>
                </div>
                
                <div style="
                    background: rgba(15, 23, 42, 0.7); 
                    border-left: 3px solid #F97316; 
                    padding: 8px 12px; 
                    border-radius: 4px; 
                    margin-top: 6px;">
                    <div style="color: #10B981; font-weight: 600; font-size: 12px; margin-bottom: 3px;">💡 Ответ Gemini:</div>
                    <div style="{hint_style} font-size: 13px;">{item['hint']}</div>
                </div>
            </div>
            """
            html_blocks.append(card_html)

        self.text_display.setHtml("".join(html_blocks))
        
        sb = self.text_display.verticalScrollBar()
        sb.setValue(sb.maximum())

    def get_gemini_hint(self, chat_id, msg_id, customer_text):
        prompt = f"""
        Ты — ведущий технический консультант и эксперт поддержки.
        Поступил запрос/вопрос: "{customer_text}"

        База знаний:
        ---
        {kb_content}
        ---

        Дай полный, развернутый и точный ответ.
        Используй списки и выделения ключевых моментов для удобства оператора.
        """
        try:
            response = client.models.generate_content(
                model=ACTIVE_MODEL,
                contents=prompt
            )
            hint = response.text.replace("\n", "<br>")
            self.update_item_signal.emit(chat_id, msg_id, customer_text, hint)
        except Exception as e:
            print(f"Ошибка Gemini ({msg_id}): {e}")
            self.update_item_signal.emit(chat_id, msg_id, customer_text, f"Ошибка API: {e}")

    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton:
            self.drag_position = event.globalPosition().toPoint() - self.frameGeometry().topLeft()
            event.accept()

    def mouseMoveEvent(self, event):
        if event.buttons() == Qt.MouseButton.LeftButton and self.drag_position is not None:
            self.move(event.globalPosition().toPoint() - self.drag_position)
            event.accept()
